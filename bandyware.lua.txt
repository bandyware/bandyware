repeat task.wait() until game:IsLoaded()

local Keybind = shared.Keybind or "RightShift"
local collectionService = game:GetService("CollectionService")
local debris = game:GetService("Debris")
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local LocalPlayer = Players.LocalPlayer
local Camera = workspace.CurrentCamera

-- Configurações do Aim Assist (Camlock)
local CamlockKey = Enum.KeyCode.CapsLock
local Smoothness = 0.22
local MaxDistance = 150
local LockAngle = 160

local CamlockEnabled = false
local Target = nil

local Icons = {
    ["iron"] = "rbxassetid://6850537969",
    ["bee"] = "rbxassetid://7343272839",
    ["natures_essence_1"] = "rbxassetid://11003449842",
    ["thorns"] = "rbxassetid://9134549615",
    ["mushrooms"] = "rbxassetid://9134534696",
    ["wild_flower"] = "rbxassetid://9134545166",
    ["crit_star"] = "rbxassetid://9866757805",
    ["vitality_star"] = "rbxassetid://9866757969"
}

local itemEspFold = Instance.new("Folder")
local playerEspFold = Instance.new("Folder")
local gui = Instance.new("ScreenGui", LocalPlayer.PlayerGui)
gui.ResetOnSpawn = false
itemEspFold.Parent = gui
playerEspFold.Parent = gui

local kitHidden = false

local function isKeybindValid(key)
    return Enum.KeyCode[key] ~= nil
end

local function showNotification(message)
    local notification = Instance.new("TextLabel")
    notification.Size = UDim2.new(0, 300, 0, 50)
    notification.Position = UDim2.new(0.5, -150, 0, 50)
    notification.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    notification.TextColor3 = Color3.new(1, 1, 1)
    notification.Font = Enum.Font.SourceSansBold
    notification.TextSize = 20
    notification.Text = message
    notification.AnchorPoint = Vector2.new(0.5, 0)
    notification.Parent = gui

    debris:AddItem(notification, 3)
end

if not isKeybindValid(Keybind) then
    showNotification("Invalid keybind! Defaulting to RightShift.")
    Keybind = "RightShift"
else
    showNotification("Press "..tostring(Keybind).." to toggle Kit ESP")
end

-- Sistema de Ícones/Kits
local itemEspObjs = {}

local function espadd(v, icon)
    if not v then return end
    local billboard = Instance.new("BillboardGui")
    billboard.Parent = itemEspFold
    billboard.Name = "ItemESP"
    billboard.StudsOffsetWorldSpace = Vector3.new(0, 3, 1.5)
    billboard.Size = UDim2.new(0, 32, 0, 32)
    billboard.AlwaysOnTop = true
    billboard.Adornee = v
    
    local image = Instance.new("ImageLabel")
    image.BackgroundTransparency = 0.5
    image.BorderSizePixel = 0
    image.Image = Icons[icon] or ""
    image.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
    image.Size = UDim2.new(0, 32, 0, 32)
    image.AnchorPoint = Vector2.new(0.5, 0.5)
    image.Parent = billboard
    
    local uicorner = Instance.new("UICorner")
    uicorner.CornerRadius = UDim.new(0, 4)
    uicorner.Parent = image
    
    itemEspObjs[v] = billboard
end

local kitConnections = {}

local function resetKits()
    for _, v in pairs(kitConnections) do
        pcall(function() v:Disconnect() end)
    end
    itemEspFold:ClearAllChildren()
    table.clear(itemEspObjs)
end

local function addKit(tag, icon, custom)
    if not custom then
        local con1 = collectionService:GetInstanceAddedSignal(tag):Connect(function(v)
            if v and v.PrimaryPart then espadd(v.PrimaryPart, icon) end
        end)
        local con2 = collectionService:GetInstanceRemovedSignal(tag):Connect(function(v)
            if v and v.PrimaryPart and itemEspObjs[v.PrimaryPart] then
                itemEspObjs[v.PrimaryPart]:Destroy()
                itemEspObjs[v.PrimaryPart] = nil
            end
        end)
        table.insert(kitConnections, con1)
        table.insert(kitConnections, con2)
        for _, v in pairs(collectionService:GetTagged(tag)) do
            if v and v.PrimaryPart then espadd(v.PrimaryPart, icon) end
        end
    else
        local function check(v)
            if v.Name == tag and v.ClassName == "Model" and v.PrimaryPart then
                espadd(v.PrimaryPart, icon)
            end
        end
        table.insert(kitConnections, workspace.ChildAdded:Connect(check))
        table.insert(kitConnections, workspace.ChildRemoved:Connect(function(v)
            pcall(function()
                if v and v.PrimaryPart and itemEspObjs[v.PrimaryPart] then
                    itemEspObjs[v.PrimaryPart]:Destroy()
                    itemEspObjs[v.PrimaryPart] = nil
                end
            end)
        end))
        for _, v in pairs(workspace:GetChildren()) do
            check(v)
        end
    end
end

local function loadKitESP()
    resetKits()
    addKit("hidden-metal", "iron")
    addKit("bee", "bee")
    addKit("treeOrb", "natures_essence_1")
    addKit("Thorns", "thorns", true)
    addKit("Mushrooms", "mushrooms", true)
    addKit("Flower", "wild_flower", true)
    addKit("CritStar", "crit_star", true)
	addKit("VitalityStar", "vitality_star", true)
end

-- ESP 3D de Jogadores (Sempre Ativado, Vermelho, Ignora Time Próprio)
local playerBoxes = {}
local permanentConnections = {}

local function createPlayerBox(player)
    if playerBoxes[player] then return end
    
    local boxHolder = Instance.new("Highlight")
    boxHolder.Name = player.Name .. "_ESP"
    boxHolder.Adornee = nil
    boxHolder.FillColor = Color3.fromRGB(255, 0, 0)
    boxHolder.FillTransparency = 0.5
    boxHolder.OutlineColor = Color3.fromRGB(255, 255, 255)
    boxHolder.OutlineTransparency = 0
    boxHolder.Parent = playerEspFold
    
    playerBoxes[player] = boxHolder

    local connection
    connection = RunService.RenderStepped:Connect(function()
        if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") or player == LocalPlayer then
            boxHolder.Adornee = nil
            return
        end

        if player.Team and LocalPlayer.Team and player.Team == LocalPlayer.Team then
            boxHolder.Adornee = nil
            return
        end

        boxHolder.Adornee = player.Character
    end)
    
    table.insert(permanentConnections, connection)
end

local function initPlayerESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            createPlayerBox(player)
        end
    end
    
    table.insert(permanentConnections, Players.PlayerAdded:Connect(function(player)
        if player ~= LocalPlayer then
            createPlayerBox(player)
        end
    end))
    
    table.insert(permanentConnections, Players.PlayerRemoving:Connect(function(player)
        if playerBoxes[player] then
            playerBoxes[player]:Destroy()
            playerBoxes[player] = nil
        end
    end))
end

-- Funções do Aim Assist (Camlock)
local function GetAngle(targetPos)
    local camDir = Camera.CFrame.LookVector
    local dirToTarget = (targetPos - Camera.CFrame.Position).Unit
    return math.deg(math.acos(camDir:Dot(dirToTarget)))
end

local function FindNearestEnemy()
    local closest = nil
    local closestDist = MaxDistance

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Team ~= LocalPlayer.Team and plr.Character then
            local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
            local humanoid = plr.Character:FindFirstChild("Humanoid")
            
            if hrp and humanoid and humanoid.Health > 0 then
                local dist = (hrp.Position - Camera.CFrame.Position).Magnitude
                local angle = GetAngle(hrp.Position)

                if dist <= closestDist and angle <= (LockAngle / 2) then
                    closestDist = dist
                    closest = hrp
                end
            end
        end
    end

    return closest
end

-- Loop Principal do Aim Assist
RunService.RenderStepped:Connect(function()
    if CamlockEnabled and Target then
        local character = Target.Parent
        local humanoid = character and character:FindFirstChild("Humanoid")

        if humanoid and humanoid.Health > 0 then
            local camPos = Camera.CFrame.Position
            local targetPos = Target.Position

            Camera.CFrame = Camera.CFrame:Lerp(
                CFrame.new(camPos, targetPos),
                Smoothness
            )
        else
            CamlockEnabled = false
            Target = nil
        end
    end
end)

-- Gerenciamento de Teclas (Keybinds)
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end

    -- Toggle do Kit ESP com a tecla principal (ex: RightShift)
    if input.KeyCode == Enum.KeyCode[Keybind] then
        if kitHidden then 
            loadKitESP() 
            kitHidden = false 
            showNotification("Kit ESP: ON")
        else 
            resetKits() 
            kitHidden = true 
            showNotification("Kit ESP: OFF")
        end
    end

    -- Toggle do Aim Assist com CapsLock
    if input.KeyCode == CamlockKey then
        if not CamlockEnabled then
            Target = FindNearestEnemy()
            if Target then
                CamlockEnabled = true
            end
        else
            CamlockEnabled = false
            Target = nil
        end
    end
end)

-- Inicialização
initPlayerESP()
loadKitESP()
